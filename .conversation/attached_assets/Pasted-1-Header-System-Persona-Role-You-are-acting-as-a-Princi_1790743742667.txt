7. User Flow
Registration & Farm Onboarding: User creates an account and specifies a farm plot (location name, area size in Hectares/Acres, soil type, irrigation capability).

Soil Baseline Entry: User enters optional or tested soil chemical parameters (N,P,K levels, pH, Organic Carbon %).

Task Selection:

Option A: Generate general growth-stage advisory.

Option B: Diagnose a suspected crop disease or pest infestation.

Option C: Calculate fertilizer dosages for a target yield.

Option D: View/export seasonal operation and irrigation schedule.

AI Processing: Express backend prepares contextual system prompts, injects stored farm metadata, validates inputs with Zod, and calls Gemini via @google/genai with strict responseSchema.

Actionable Delivery: UI renders high-contrast, structured response cards complete with dosage meters, warning badges, step-by-step actions, and chemical safety instructions.

Execution & Feedback Loop: User marks actions as completed and submits outcome ratings (1-5 stars) to refine future recommendations.

10. Database Schema (Production SQL for PostgreSQL)
SQL
-- Enable UUID Extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Enum Definitions
CREATE TYPE user_role AS ENUM ('farmer', 'agronomist', 'extension_worker', 'admin');
CREATE TYPE severity_level AS ENUM ('low', 'moderate', 'high', 'critical');
CREATE TYPE advisory_status AS ENUM ('draft', 'active', 'completed', 'archived');

-- 1. Users Table
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    role user_role DEFAULT 'farmer',
    preferred_language VARCHAR(10) DEFAULT 'en',
    phone_number VARCHAR(20),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. Farms Table
CREATE TABLE farms (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    location_name VARCHAR(150) NOT NULL,
    latitude DECIMAL(10, 8),
    longitude DECIMAL(11, 8),
    total_area DECIMAL(10, 2) NOT NULL,
    area_unit VARCHAR(20) DEFAULT 'Hectares',
    soil_type VARCHAR(50) NOT NULL,
    irrigation_method VARCHAR(50) NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Soil Profiles Table
CREATE TABLE soil_profiles (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    ph_level DECIMAL(4, 2),
    nitrogen_kg_ha DECIMAL(8, 2),
    phosphorus_kg_ha DECIMAL(8, 2),
    potassium_kg_ha DECIMAL(8, 2),
    organic_carbon_pct DECIMAL(4, 2),
    electrical_conductivity DECIMAL(5, 2),
    tested_at DATE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. Crops Table
CREATE TABLE crops (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    crop_name VARCHAR(100) NOT NULL,
    variety VARCHAR(100),
    sowing_date DATE NOT NULL,
    current_stage VARCHAR(50) NOT NULL,
    expected_harvest_date DATE,
    target_yield_per_unit DECIMAL(10, 2),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. Advisories Table
CREATE TABLE advisories (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    crop_id UUID REFERENCES crops(id) ON DELETE SET NULL,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(200) NOT NULL,
    category VARCHAR(50) NOT NULL,
    summary TEXT NOT NULL,
    full_payload JSONB NOT NULL,
    status advisory_status DEFAULT 'active',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 6. Diagnostics Table
CREATE TABLE diagnostics (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    crop_id UUID REFERENCES crops(id) ON DELETE SET NULL,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    image_url TEXT,
    symptoms_description TEXT NOT NULL,
    diagnosis_name VARCHAR(150) NOT NULL,
    scientific_name VARCHAR(150),
    confidence_score DECIMAL(5, 2) NOT NULL,
    severity severity_level NOT NULL,
    diagnostic_payload JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 7. Fertilizer Plans Table
CREATE TABLE fertilizer_plans (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    farm_id UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
    crop_id UUID NOT NULL REFERENCES crops(id) ON DELETE CASCADE,
    target_yield DECIMAL(10, 2) NOT NULL,
    n_deficit DECIMAL(8, 2) NOT NULL,
    p_deficit DECIMAL(8, 2) NOT NULL,
    k_deficit DECIMAL(8, 2) NOT NULL,
    recommendations_payload JSONB NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 8. Advisory Feedback Table
CREATE TABLE advisory_feedback (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    advisory_id UUID NOT NULL REFERENCES advisories(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    rating INT CHECK (rating >= 1 AND rating <= 5),
    was_effective BOOLEAN DEFAULT TRUE,
    comments TEXT,
    actual_yield_impact VARCHAR(100),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Performance Indexes

-- Helper Function to extract current session user UUID
CREATE OR REPLACE FUNCTION current_app_user_id() RETURNS UUID AS $$
    SELECT NULLIF(current_setting('app.current_user_id', true), '')::UUID;
$$ LANGUAGE sql STABLE;

-- Farms Policy
CREATE POLICY farm_user_isolation ON farms
    FOR ALL
    USING (user_id = current_app_user_id());

-- Soil Profiles Policy (via Farm link)

    FOR ALL
    USING (farm_id IN (SELECT id FROM farms WHERE user_id = current_app_user_id()));

import { Type } from '@google/genai';
import { ai, GEMINI_MODEL_FAST } from '../config/gemini.js';

export async function generateCropAdvisory(params: {
  cropName: string;
  stage: string;
  soilType: string;
  location: string;
  observedIssues?: string;
}) {
  const prompt = `
Generate a comprehensive crop advisory for the following farm profile:
- Crop: ${params.cropName}
- Current Growth Stage: ${params.stage}
- Soil Type: ${params.soilType}
- Location: ${params.location}
- Observed Field Issues/Notes: ${params.observedIssues || 'None reported'}

Return a structured advisory with immediate actions, nutrient recommendations, irrigation guidance, and weather/disease risks.
`;

  const response = await ai.models.generateContent({
    model: GEMINI_MODEL_FAST,
    contents: prompt,
    config: {
      systemInstruction: 'You are AgriAdvisor-Core. Return strict valid JSON.',
      responseMimeType: 'application/json',
      responseSchema: {
        type: Type.OBJECT,
        properties: {
          advisoryTitle: { type: Type.STRING },
          growthStageSummary: { type: Type.STRING },
          immediateActionItems: {
            type: Type.ARRAY,
            items: {
              type: Type.OBJECT,

                priority: { type: Type.STRING, enum: ['High', 'Medium', 'Low'] },

        ]
      }
    }
  });

  return JSON.parse(response.text!);
}

18. Security Requirements
JWT Authentication & Password Hashing: Passwords must be hashed using bcryptjs with a minimum cost factor of 10. Passwords must never be returned in API payloads.

Strict File Upload Validation: The /api/diagnostics/analyze endpoint accepts only image/jpeg, image/png, and image/webp MIME types with a maximum file limit of 5MB. Files are held in memory buffers (multer.memoryStorage()) and wiped after AI SDK transmission.

CORS & Rate Limiting:

Limit API endpoints using express-rate-limit (Max 100 requests per 15-minute window for standard endpoints; max 10 requests per minute for Gemini AI generation endpoints).

Configure CORS to restrict requests strictly to the application origin.

Input Sanitization: All incoming text payloads must pass through Zod parsing and HTML stripping to prevent SQL Injection and Cross-Site Scripting (XSS).

API Key Isolation: Ensure GEMINI_API_KEY and database credentials exist exclusively in server environment variables.
